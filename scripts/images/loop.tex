% The two-timescale loop (fig:loop). TikZ replacement for loop.py.
% Build:  latexmk -pdf loop.tex  -> loop.pdf, copy to Images/loop.pdf.
% Fonts match the thesis (lmodern + T1, see styles/myreport.sty).
% Two portrait panels side by side: (a) the F_fix baseline first, then
% (b) the adaptive model, so the reader sees the feedback path being added.
\documentclass[border=2pt]{standalone}
\usepackage{lmodern}
\usepackage[T1]{fontenc}
\usepackage{amsmath}
\usepackage{tikz}
\usetikzlibrary{positioning,fit,backgrounds,arrows.meta,calc}

% Palette shared with base.py (C dict).
\definecolor{storeF}{HTML}{CFE3F7}
\definecolor{taskF}{HTML}{F6E7C1}
\definecolor{ctrlF}{HTML}{D7F0D8}
\definecolor{estF}{HTML}{F7D6D6}
\definecolor{ink}{HTML}{43505C}
\definecolor{fb}{HTML}{C0392B}
\definecolor{muted}{HTML}{5A6673}

\tikzset{
  box/.style={draw=ink, line width=0.8pt, rounded corners=3pt, align=center,
              minimum width=4.0cm, minimum height=1.55cm, inner sep=4pt,
              font=\small},
  flow/.style={-{Stealth[length=2.6mm]}, line width=1pt, draw=ink},
  slow/.style={-{Stealth[length=2.6mm]}, line width=1.1pt, draw=fb},
  lbl/.style={font=\small, text=fb, fill=white, inner sep=2pt},
  fastzone/.style={draw=muted, dashed, rounded corners=6pt, inner sep=7pt},
}

% #1 = feedback (1) or F_fix (0), #2 = panel title
\newcommand{\looppanel}[2]{%
\begin{tikzpicture}
  \ifnum#1=1
    \node[box, fill=storeF] (S) {\textbf{Storage layer} $F$\\
      slow, every $\Delta$ steps\\
      $x_t = F(x_{t-\Delta}, \hat\rho_t, \hat\mu_t, \hat w_t)$};
  \else
    \node[box, fill=storeF] (S) {\textbf{Storage layer} $F_{\mathrm{fix}}$\\
      $F_{\mathrm{fix}}(x,\cdot) = x$\\
      ignores the estimates};
  \fi
  \node[box, fill=taskF, below=0.9cm of S] (T) {\textbf{Task generation}\\
    $\tau_j = (t_j^{\mathrm{released}}, s_j, g_j, k_j)$\\ pickup $s_j$ holds $k_j$ under $x_t$};
  \node[box, fill=ctrlF, below=0.9cm of T] (P) {\textbf{MAPD controller} $\pi$\\
    fast, every timestep\\ assignment and routing};
  \node[box, fill=estF, minimum width=3.3cm, right=1.9cm of P] (E)
    {\textbf{Causal estimates}\\ demand $\hat\rho_t(k)$\\
     traffic $\hat\mu_t(e),\ \hat w_t(e)$};

  \begin{scope}[on background layer]
    \node[fastzone, fit=(T)(P)] (Z) {};
  \end{scope}
  \node[font=\footnotesize, text=muted, anchor=north west, align=left]
    at (Z.south west) {fast loop, one environment timestep};

  \draw[flow] (S) -- node[right, font=\footnotesize] {$x_t$} (T);
  \draw[flow] (T) -- node[right, font=\footnotesize] {$\tau_j$} (P);

  \draw[slow] (P.east) -- node[lbl, above=2pt] {measure} (E.west);
  \ifnum#1=1
    \draw[slow] (E.north) |- node[lbl, pos=0.3, align=center]
      {feedback,\\ every $\Delta$} (S.east);
  \else
    % Genuine absence, not a ghosted arrow: under F_fix nothing connects
    % the estimate to storage.
    \node[font=\small\itshape, text=muted, align=center]
      at (E.north |- T) {no feedback,\\ $x_t$ held fixed};
    % Invisible path with the same extent as the feedback arrow, so both
    % panels get identical bounding boxes and line up side by side.
    \path (E.north) |- (S.east);
  \fi

  \node[anchor=south west, font=\normalsize]
    at ($(S.north west)+(0,0.35)$) {#2};
\end{tikzpicture}}

\begin{document}
\begin{tabular}{@{}c@{\hspace{0.6cm}}c@{}}
  \looppanel{0}{(a) $F_{\mathrm{fix}}$ baseline, no feedback} &
  \looppanel{1}{(b) Adaptive storage, feedback closes the loop}
\end{tabular}
\end{document}
